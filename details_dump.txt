        async function extractBase64(file) {
            return new Promise((resolve) => {
                let reader = new FileReader();
                reader.onloadend = () => resolve(reader.result.split(',')[1]);
                reader.readAsDataURL(file);
            });
        }

        let uploadedImageB64 = null;
        
        function handleImage(e) {
            if(e.target.files[0]) {
                extractBase64(e.target.files[0]).then(b => {
                    uploadedImageB64 = b;
                    let preview = document.getElementById('imgPreview');
                    preview.src = "data:image/jpeg;base64," + b;
                    preview.style.display = 'block';
                });
            }
        }

        function startMic() {
            let SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition;
            if (!SpeechRecognition) return alert("Microphone not supported in this browser.");
            let rec = new SpeechRecognition();
            rec.lang = currentLang === 'te' ? 'te-IN' : (currentLang === 'hi' ? 'hi-IN' : 'en-US');
            rec.onstart = () => document.getElementById('diseaseInput').placeholder = "Listening...";
            rec.onresult = (e) => {
                document.getElementById('diseaseInput').value = e.results[0][0].transcript;
                askDisease();
            };
            rec.start();
        }

        async function askDisease() {
            document.getElementById('loadingIndicator').style.display = 'block';
            let query = document.getElementById('diseaseInput').value || "What disease is in this image and how to cure it?";

            let prompt = "You are AXIOM. The user wants to detect a crop disease. Language: " + currentLang +
                ". User query: " + query + ". Give the disease name, causes, required chemicals to cure it, and few main points. " +
                "Output entirely in " + currentLang + " using plain text with no markdown symbols. Use line breaks.";

            let bodyData = { contents: [{ role: "user", parts: [{ text: prompt }] }] };

            if (uploadedImageB64) {
                bodyData.contents[0].parts.push({
                    inline_data: { mime_type: "image/jpeg", data: uploadedImageB64 }
                });
            }

            try {
                let apiKey = GEMINI_API_KEY;
                let res = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-3.1-flash-lite:generateContent?key=${apiKey}`, {
                    method: 'POST',
                    headers: {'Content-Type':'application/json'},
                    body: JSON.stringify(bodyData)
                });
                let data = await res.json();
                let reply = data.candidates[0].content.parts[0].text;
                document.getElementById('loadingIndicator').style.display = 'none';
                
                let out = document.getElementById('cardsContainer');
                out.innerHTML = `<div class="disease-cure-box">${reply.replace(/\\n/g, '<br>')}</div>`;
                
                // Disease completely silenced.
                
            } catch(e) {
                document.getElementById('loadingIndicator').innerText = "Network Error!";
            }
        }

                         <div class="card-header"><div class="card-icon">${item.icon}</div><h2 class="card-title">${data.title}</h2></div>
                         <p class="card-desc">${data.desc}</p>
                         <a href="${data.url}" class="card-link" target="_blank">${data.linkText} →</a>
                     </div>`;
                 });
                 
                 document.getElementById('cardsContainer').innerHTML = `
                   <div class="disease-cure-box" style="border-color:#fbbf24; margin-bottom: 30px;">
                       <h2 style="margin-top:0; color:#fbbf24;">📈 Market Prices Today</h2>
                       ${reply.replace(/\n/g, '<br>')}
                   </div>
                   ${marketLinks}
                 `;
                 // Silenced fallback
             } catch(e) {
                 document.getElementById('loadingIndicator').innerText = "Market Network Error!";
             }
        }

        async function generateWeatherSuggestions(lat, lon) {
             try {
                 let res = await fetch(`https://api.open-meteo.com/v1/forecast?latitude=${lat}&longitude=${lon}&current_weather=true`);
                 let data = await res.json();
                 let temp = data.current_weather.temperature;
                 let wc = data.current_weather.weathercode;
                 let wName = wc <= 3 ? "Clear" : wc <= 48 ? "Cloudy" : wc <= 67 ? "Rainy" : "Stormy";
                 
                 // Reverse geocode to get city name for better realism
                 let city = "";
                 try {
                     let geoRes = await fetch(`https://nominatim.openstreetmap.org/reverse?format=json&lat=${lat}&lon=${lon}`);
                     let geoData = await geoRes.json();
                     city = geoData.address.city || geoData.address.town || geoData.address.state || "your area";
                 } catch(e) { city = "your local farming zone"; }

                 
                 let prompt = "You are AXIOM. The user is a farmer. Current accurate local weather for " + city + " is " + temp + "°C and " + wName + 
                              ". Based on this STRICTLY, give exactly 3 bullet points of smart crop or farming suggestions to maximize yield. " +
                              "Output ONLY in the requested language: " + currentLang + ". DO NOT USE MARKDOWN. Just text sentences.";

                 let apiKey = GEMINI_API_KEY;
                 let geminiRes = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-3.1-flash-lite:generateContent?key=${apiKey}`, {
                     method: 'POST',
                     headers: {'Content-Type':'application/json'},
                     body: JSON.stringify({ contents: [{ role: "user", parts: [{ text: prompt }] }] })
                 });
                 let geminiData = await geminiRes.json();
                 let reply = geminiData.candidates[0].content.parts[0].text;
                 
                 document.getElementById('loadingIndicator').style.display = 'none';
                 document.getElementById('cardsContainer').innerHTML = `
                   <div class="disease-cure-box">
                       <h2 style="margin-top:0">⛅ ${temp}°C (${wName})</h2>
                       ${reply.replace(/\\n/g, '<br>')}
                   </div>
                 `;
                 speakText(reply.replace(/\\*/g, ''));
             } catch(e) {
                 document.getElementById('loadingIndicator').innerText = "Network Error!";
             }
        }
